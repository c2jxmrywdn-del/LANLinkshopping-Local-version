# 统一日志轮转，避免长期运行把宿主磁盘写满（本地开发稳定性）。
x-logging: &default-logging
  driver: json-file
  options:
    max-size: "20m"
    max-file: "3"

services:
  mysql:
    image: mysql:8.4
    logging: *default-logging
    # 24GB 宿主机 / 8-12GB WSL：限制单服务内存，避免缓冲池/连接把 WSL 内存吃满
    mem_limit: 1g
    environment:
      MYSQL_DATABASE: ${MYSQL_DATABASE:-lanlink_shopping}
      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD:?Set MYSQL_ROOT_PASSWORD in .env}
    command: --character-set-server=utf8mb4 --collation-server=utf8mb4_unicode_ci
    volumes:
      - mysql-data:/var/lib/mysql
      - ./backend/src/main/resources/sql:/docker-entrypoint-initdb.d:ro
    healthcheck:
      test: ["CMD-SHELL", "mysqladmin ping -h 127.0.0.1 -uroot -p$${MYSQL_ROOT_PASSWORD} --silent"]
      interval: 10s
      timeout: 5s
      retries: 12
    restart: unless-stopped

  mailpit:
    image: axllent/mailpi
